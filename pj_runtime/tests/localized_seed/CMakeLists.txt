if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.22)
  project(pj_localized_seed_tests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Core Test)
endif()
add_executable(localized_seed_test ../localized_seed_test.cpp)
set_target_properties(localized_seed_test PROPERTIES AUTOMOC ON WIN32_EXECUTABLE FALSE)
target_include_directories(localized_seed_test PRIVATE ../../include)
target_link_libraries(localized_seed_test PRIVATE Qt6::Core Qt6::Test)
if(MSVC)
  target_compile_options(localized_seed_test PRIVATE /utf-8 /W4 /WX)
else()
  target_compile_options(localized_seed_test PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
endif()
add_test(NAME LocalizedPluginSeedTest COMMAND localized_seed_test)
