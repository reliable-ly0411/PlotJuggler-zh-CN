cmake_minimum_required(VERSION 3.21)
project(PjRegistryFixtureSmoke LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This verifies real Win32 registry semantics; Windows is required")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
if(NOT EXISTS "${PJ_TEST_MAINS_DIR}/pj_test_windows_registry.h")
  message(FATAL_ERROR "Set PJ_TEST_MAINS_DIR to the patched source's cmake/test_mains")
endif()
add_executable(registry_fixture_smoke main.cpp)
target_compile_features(registry_fixture_smoke PRIVATE cxx_std_20)
target_include_directories(registry_fixture_smoke PRIVATE "${PJ_TEST_MAINS_DIR}")
target_link_libraries(registry_fixture_smoke PRIVATE Qt6::Core advapi32)
