cmake_minimum_required(VERSION 3.22)
project(pj_native_ci_temp_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/PjVersions.cmake")
pj_read_versions("${CMAKE_CURRENT_LIST_DIR}/../../../versions.env")
find_package(Qt6 ${PJ_QT_VERSION} REQUIRED COMPONENTS Core)
add_executable(pj_native_ci_temp_probe main.cpp)
target_link_libraries(pj_native_ci_temp_probe PRIVATE Qt6::Core)
if(WIN32)
    target_link_libraries(pj_native_ci_temp_probe PRIVATE Advapi32)
endif()
if(MSVC)
    target_compile_options(pj_native_ci_temp_probe PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(pj_native_ci_temp_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
