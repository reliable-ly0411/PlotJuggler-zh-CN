cmake_minimum_required(VERSION 3.21)
project(native_app_marker_probe LANGUAGES CXX)

# The caller may supply the version when packaging this directory separately.
if(NOT DEFINED PJ_QT_VERSION)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../../versions.env" version_line REGEX "^PJ_QT_VERSION=")
  string(REGEX REPLACE "^PJ_QT_VERSION=" "" PJ_QT_VERSION "${version_line}")
endif()
if(NOT PJ_QT_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "A pinned PJ_QT_VERSION is required")
endif()
find_package(Qt6 ${PJ_QT_VERSION} EXACT REQUIRED COMPONENTS Core)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(marker_probe_policy STATIC policy.cpp)
target_link_libraries(marker_probe_policy PUBLIC Qt6::Core)
target_include_directories(marker_probe_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(native_app_marker_probe main.cpp)
target_link_libraries(native_app_marker_probe PRIVATE marker_probe_policy Qt6::Core)
target_compile_definitions(native_app_marker_probe PRIVATE PROBE_QT_VERSION="${PJ_QT_VERSION}")
if(WIN32)
  target_link_libraries(native_app_marker_probe PRIVATE advapi32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(marker_probe_policy_test tests/policy_test.cpp)
  target_link_libraries(marker_probe_policy_test PRIVATE marker_probe_policy Qt6::Core)
  add_test(NAME marker_probe_policy COMMAND marker_probe_policy_test)
  if(NOT WIN32)
    add_test(NAME marker_probe_unsupported COMMAND marker_probe_policy_test $<TARGET_FILE:native_app_marker_probe>)
  else()
    add_executable(plotjuggler_marker_test_fixture tests/known_process_fixture.cpp)
    add_executable(marker_probe_windows_tests tests/windows_probe_test.cpp)
    target_link_libraries(marker_probe_windows_tests PRIVATE marker_probe_policy Qt6::Core advapi32)
    target_compile_definitions(marker_probe_windows_tests PRIVATE PROBE_QT_VERSION="${PJ_QT_VERSION}")
    add_dependencies(marker_probe_windows_tests plotjuggler_marker_test_fixture native_app_marker_probe)
    foreach(case IN ITEMS identity qt cli marker_paths access_denied processes)
      add_test(NAME marker_probe_windows_${case} COMMAND marker_probe_windows_tests ${case})
      set_tests_properties(marker_probe_windows_${case} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    endforeach()
  endif()
endif()
