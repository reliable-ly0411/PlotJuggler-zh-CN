cmake_minimum_required(VERSION 3.22)
project(ResourceImageTools LANGUAGES CXX)

set(PJ_IMAGE_TOOLS_QT_MAJOR 6 CACHE STRING "Qt major for the isolated Core/Gui build")
set(CMAKE_CXX_STANDARD 20)
if(PJ_IMAGE_TOOLS_QT_MAJOR STREQUAL "5")
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt${PJ_IMAGE_TOOLS_QT_MAJOR} REQUIRED COMPONENTS Core Gui)
get_filename_component(PJ_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(QT_GUI Qt${PJ_IMAGE_TOOLS_QT_MAJOR}::Gui)

enable_testing()
add_executable(resource_image_smoke
  ../resource_image_test.cpp
  ${PJ_SOURCE_ROOT}/plotjuggler_app/resource_image.cpp)
target_include_directories(resource_image_smoke PRIVATE ${PJ_SOURCE_ROOT}/plotjuggler_app)
target_link_libraries(resource_image_smoke PRIVATE ${QT_GUI})
add_test(NAME GenericResourceImage COMMAND resource_image_smoke)
set_tests_properties(GenericResourceImage PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

add_executable(scaling_probe scaling_probe.cpp
  ${PJ_SOURCE_ROOT}/plotjuggler_app/resource_image.cpp)
target_include_directories(scaling_probe PRIVATE ${PJ_SOURCE_ROOT}/plotjuggler_app)
target_link_libraries(scaling_probe PRIVATE ${QT_GUI})

add_executable(hash_buffer_probe hash_probe.cpp)
target_link_libraries(hash_buffer_probe PRIVATE Qt${PJ_IMAGE_TOOLS_QT_MAJOR}::Core)
add_executable(hash_image_probe hash_probe.cpp)
target_compile_definitions(hash_image_probe PRIVATE PJ_HASH_IMAGES=1)
target_link_libraries(hash_image_probe PRIVATE ${QT_GUI})

option(PJ_BUILD_LOCALIZED_MEME_DIAGNOSTIC "Build the exact full-catalog scaling regression" OFF)
if(PJ_BUILD_LOCALIZED_MEME_DIAGNOSTIC)
  find_package(Qt${PJ_IMAGE_TOOLS_QT_MAJOR} REQUIRED COMPONENTS Widgets)
  set(meme_root "${PJ_SOURCE_ROOT}/plotjuggler_app/resources/memes")
  file(GLOB originals "${meme_root}/meme_*.jpg")
  file(GLOB_RECURSE localized "${meme_root}/zh_CN/*.png")
  set(meme_files ${originals} ${localized} "${meme_root}/zh_CN.json")
  set(qrc "${CMAKE_CURRENT_BINARY_DIR}/diagnostic_memes.qrc")
  file(WRITE "${qrc}" "<RCC><qresource prefix=\"/resources/memes\">\n")
  foreach(path IN LISTS meme_files)
    file(RELATIVE_PATH alias "${meme_root}" "${path}")
    file(APPEND "${qrc}" "<file alias=\"${alias}\">${path}</file>\n")
  endforeach()
  file(APPEND "${qrc}" "</qresource></RCC>\n")
  qt5_add_big_resources(meme_resources "${qrc}")
  set_source_files_properties(${meme_resources} PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
  add_custom_target(diagnostic_meme_resources DEPENDS ${meme_resources})
  add_executable(meme_scaling_regression ../localized_meme_test.cpp
    ${PJ_SOURCE_ROOT}/plotjuggler_app/localized_meme.cpp
    ${PJ_SOURCE_ROOT}/plotjuggler_app/resource_image.cpp ${meme_resources})
  add_dependencies(meme_scaling_regression diagnostic_meme_resources)
  target_include_directories(meme_scaling_regression PRIVATE "${PJ_SOURCE_ROOT}" "${PJ_SOURCE_ROOT}/plotjuggler_app")
  target_link_libraries(meme_scaling_regression PRIVATE Qt${PJ_IMAGE_TOOLS_QT_MAJOR}::Widgets)
endif()

option(PJ_BUILD_IMAGE_BENCHMARKS "Build isolated, non-shipping image timing tools" OFF)
set(PJ_IMAGE_INPUTS "" CACHE PATH "Output of prepare_image_benchmarks.py")
if(PJ_BUILD_IMAGE_BENCHMARKS)
  if(NOT PJ_IMAGE_TOOLS_QT_MAJOR STREQUAL "5")
    message(FATAL_ERROR "Production comparison tools deliberately require Qt5")
  endif()
  foreach(label zh3 zh4)
    set(input "${PJ_IMAGE_INPUTS}/${label}")
    qt5_add_big_resources(resource_${label} "${input}/images_${label}.qrc")
    set_source_files_properties(${resource_${label}} PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
    add_custom_target(resources_${label} DEPENDS ${resource_${label}})
    add_executable(image_bench_${label} image_benchmark.cpp
      "${input}/localized_meme.cpp" ${resource_${label}})
    add_dependencies(image_bench_${label} resources_${label})
    target_include_directories(image_bench_${label} PRIVATE "${input}")
    target_link_libraries(image_bench_${label} PRIVATE ${QT_GUI})
    target_compile_definitions(image_bench_${label} PRIVATE PJ_BENCH_LABEL="${label}")
    if(label STREQUAL "zh4")
      target_sources(image_bench_${label} PRIVATE ${PJ_SOURCE_ROOT}/plotjuggler_app/resource_image.cpp)
      target_include_directories(image_bench_${label} PRIVATE ${PJ_SOURCE_ROOT}/plotjuggler_app)
    endif()
    if(WIN32)
      target_link_libraries(image_bench_${label} PRIVATE psapi)
    endif()
  endforeach()
endif()
