if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.22)
  project(pj_marketplace_catalog_tests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
  include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/PjVersions.cmake")
  pj_read_versions("${CMAKE_CURRENT_LIST_DIR}/../../../versions.env")
  find_package(Qt6 ${PJ_QT_VERSION} REQUIRED COMPONENTS Core Test LinguistTools)
  include("${CMAKE_CURRENT_LIST_DIR}/../../../pj_app/cmake/PjAppLanguage.cmake")
  pj_add_app_language()
endif()
add_executable(marketplace_catalog_test ../marketplace_catalog_test.cpp)
set_target_properties(marketplace_catalog_test PROPERTIES AUTOMOC ON WIN32_EXECUTABLE FALSE)
target_include_directories(marketplace_catalog_test PRIVATE ../../include ../../../pj_widgets/include)
target_link_libraries(marketplace_catalog_test PRIVATE pj_app_language Qt6::Test)
target_compile_definitions(marketplace_catalog_test PRIVATE
  PJ_REGISTRY_LOCK="${CMAKE_CURRENT_LIST_DIR}/../../../packaging/localization/registry-lock.json")
if(MSVC)
  target_compile_options(marketplace_catalog_test PRIVATE /utf-8 /W4 /WX)
else()
  target_compile_options(marketplace_catalog_test PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
endif()
add_test(NAME MarketplaceCatalogTest COMMAND marketplace_catalog_test)
